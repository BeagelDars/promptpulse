Immediate Drop-in Setup (cURL / Python / Node.js)

1. Test with cURL (Run in your terminal right now!)

curl -X POST http://localhost:8080/v1/chat/completions \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer pp_live_default" \
  -d '{
    "model": "gpt-4o",
    "messages": [{"role": "user", "content": "Hello PromptPulse!"}]
  }'

2. Python (OpenAI Client)

import openai

# Point OpenAI directly at your PromptPulse gateway
client = openai.OpenAI(
    base_url="http://localhost:8080/v1",
    api_key="pp_live_default"  # Any active PromptPulse key
)

response = client.chat.completions.create(
    model="gpt-4o",
    messages=[{"role": "user", "content": "Analyze quarterly customer retention."}]
)
print(response.choices[0].message.content)

3. Node.js / TypeScript

import OpenAI from 'openai';

const openai = new OpenAI({
  baseURL: 'http://localhost:8080/v1',
  apiKey: 'pp_live_default',
});

const completion = await openai.chat.completions.create({
  model: 'gpt-4o',
  messages: [{ role: 'user', content: 'Generate high-converting landing page copy.' }],
});
console.log(completion.choices[0].message.content);
